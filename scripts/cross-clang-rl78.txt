[binaries]
# Ensure that $PATH contains clang, the LLVM binutils and run-rl78 (rl78sim)
c = ['clang', '-target', 'rl78-unknown-elf', '-nostdlib', '-nostdlibinc']
c_ld = 'lld'
ar = 'llvm-ar'
nm = 'llvm-nm'
strip = 'llvm-strip'
exe_wrapper = ['env', 'run-rl78']

[host_machine]
system = 'none'
cpu_family = 'rl78'
cpu = 'rl78'
endian = 'little'

[properties]
skip_sanity_check = true
librt = ['-lclang_rt.builtins']
# The library works with either mirror source and ROM model; the
# application's objects say which one the program uses.
lib_c_args = ['-mmirror-source-common', '-mcommon-rom']
default_alignment = 2
# The layout RL78 simulators implement: 32 KiB of RAM below the saddr
# area; the linker script's default mirror, F1000H-F7FFFH, lies below it.
default_flash_addr = '0x000000d8'
default_flash_size = '0x0007ff28'
default_ram_addr   = '0x000f8000'
default_ram_size   = '0x00007e20'
default_stack_size = '0x00000400'
